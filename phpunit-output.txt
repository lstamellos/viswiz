  function relationConstraintMessages(relationType, fromNode, toNode) {\n
    const schema = cfg.relationTypes?.[relationType];\n
    if (!schema) return [];\n
    const messages = [];\n
    const nodeTypeLabel = (type) => cfg.nodeTypes?.[type]?.label || type;\n

    }\n
    if (fromNode && schema.source_subtype && fromNode.node_subtype && schema.source_subtype !== fromNode.node_subtype) {\n

    if (toNode && schema.target_type && toNode.node_type && schema.target_type !== toNode.node_type) {\n
      messages.push(sprintf(__('Target should be %1$s; selected %2$s.', 'viswiz'), nodeTypeLabel(schema.target_type), nodeTypeLabel(toNode.node_type)));\n
    }\n
    if (toNode && schema.target_subtype && toNode.node_subtype && schema.target_subtype !== toNode.node_subtype) {\n

  }\n

  function relationConstraintMessages(relationType, fromNode, toNode) {\n
    const schema = cfg.relationTypes?.[relationType];\n
    if (!schema) return [];\n
    const messages = [];\n
    const nodeTypeLabel = (type) => cfg.nodeTypes?.[type]?.label || type;\n
    const subtypeLabel = (type, subtype) => cfg.nodeTypes?.[type]?.subtypes?.[subtype] || subtype;\n
    if (fromNode && schema.source_type && fromNode.node_type && schema.source_type !== fromNode.node_type) {\n
      messages.push(sprintf(__('Source should be %1$s; selected %2$s.', 'viswiz'), nodeTypeLabel(schema.source_type), nodeTypeLabel(fromNode.node_type)));\n
    }\n
    if (fromNode && schema.source_subtype && fromNode.node_subtype && schema.source_subtype !== fromNode.node_subtype) {\n
      messages.push(sprintf(__('Source subtype should be %1$s; selected %2$s.', 'viswiz'), subtypeLabel(schema.source_type, schema.source_subtype), subtypeLabel(fromNode.node_type, fromNode.node_subtype)));\n
    }\n
    if (toNode && schema.target_type && toNode.node_type && schema.target_type !== toNode.node_type) {\n
      messages.push(sprintf(__('Target should be %1$s; selected %2$s.', 'viswiz'), nodeTypeLabel(schema.target_type), nodeTypeLabel(toNode.node_type)));\n
    }\n
    if (toNode && schema.target_subtype && toNode.node_subtype && schema.target_subtype !== toNode.node_subtype) {\n
      messages.push(sprintf(__('Target subtype should be %1$s; selected %2$s.', 'viswiz'), subtypeLabel(schema.target_type, schema.target_subtype), subtypeLabel(toNode.node_type, toNode.node_subtype)));\n
    }\n
    return messages;\n
  }\n

  function graphLayout(nodes, relations, renderer, width, height) {\n
    const result = new Map();\n
    if (!nodes.length) return result;\n
    if (renderer === 'org_chart' || renderer === 'flow_diagram') {\n
      const incoming = new Map(nodes.map((n) => [n.uuid, 0]));\n
      const outgoing = new Map(nodes.map((n) => [n.uuid, []]));\n
      relations.forEach((r) => {\n
        if (incoming.has(r.to_node_uuid) && outgoing.has(r.from_node_uuid)) {\n
          incoming.set(r.to_node_uuid, incoming.get(r.to_node_uuid) + 1);\n
          outgoing.get(r.from_node_uuid).push(r.to_node_uuid);\n
        }\n
      });\n
      const queue = [...nodes.filter((n) => incoming.get(n.uuid) === 0).map((n) => n.uuid)];\n
      if (!queue.length) queue.push(nodes[0].uuid);\n
      const level = new Map(queue.map((id) => [id, 0]));\n
      for (let qi = 0; qi < queue.length; qi++) {\n
        const id = queue[qi], l = level.get(id) || 0;\n
        (outgoing.get(id) || []).forEach((next) => {\n
          const nl = Math.max(level.get(next) || 0, l + 1);\n
          if (!level.has(next) || nl > level.get(next)) level.set(next, nl);\n
          if (!queue.includes(next)) queue.push(next);\n
        });\n
      }\n
      nodes.forEach((n) => { if (!level.has(n.uuid)) level.set(n.uuid, 0); });\n
      const groups = new Map(); nodes.forEach((n) => { const l = level.get(n.uuid); if (!groups.has(l)) groups.set(l, []); groups.get(l).push(n); });\n
      const maxLevel = Math.max(0, ...groups.keys());\n
      groups.forEach((group, l) => group.forEach((node, i) => {\n
        const horizontal = renderer === 'flow_diagram';\n
        const primary = (l + 1) / (maxLevel + 2);\n
        const secondary = (i + 1) / (group.length + 1);\n
        result.set(node.uuid, horizontal ? { x: width * primary, y: height * secondary } : { x: width * secondary, y: height * primary });\n
      }));\n
      return result;\n
    }\n
    const radius = Math.min(width, height) * 0.36, cx = width / 2, cy = height / 2;\n
    nodes.forEach((node, i) => {\n
      const a = (i / nodes.length) * Math.PI * 2 - Math.PI / 2;\n
      const ring = nodes.length > 30 ? 0.58 + (i % 3) * 0.2 : 1;\n
      result.set(node.uuid, { x: cx + Math.cos(a) * radius * ring, y: cy + Math.sin(a) * radius * ring });\n
    });\n
    return result;\n
  }\n

  function renderGraph(container, spec) {\n

    let query = '';\n

\n

      view.x += (view.w - nextW) / 2;\n
      view.y += (view.h - nextH) / 2;\n
      view.w = nextW;\n
      view.h = nextH;\n
      applyView();\n
    };\n
    const resetView = () => {\n

      view = { ...baseView };\n
      applyView();\n
    };\n

      const toolbar = el('div', { class: 'viswiz-graph-toolbar' });\n

      const zoomIn = el('button', { type: 'button', class: 'viswiz-graph-tool', title: __('Zoom in', 'viswiz'), 'aria-label': __('Zoom in', 'viswiz') }, '+');\n
      status = el('span', { class: 'viswiz-graph-status', 'aria-live': 'polite' });\n
      if (spec.settings?.show_graph_search !== false) toolbar.appendChild(search);\n

      if (spec.settings?.show_graph_zoom !== false) toolbar.append(zoomOut, zoomReset, zoomIn);\n

    function draw() {\n
      stage.replaceChildren();\n
      const nodes = allNodes.filter((node) => {\n
        if (nodeType && node.node_type !== nodeType) return false;\n
        if (!query) return true;\n

      });\n

        stage.appendChild(el('p', { class: 'viswiz-empty' }, __('No matching nodes', 'viswiz')));\n
        activeSvg = null;\n
        return;\n
      }\n
      const ids = new Set(nodes.map((n) => n.uuid));\n

      if (status) status.textContent = `${nodes.length}/${allNodes.length} ${__('nodes', 'viswiz')} · ${relations.length}/${allRelations.length} ${__('relations', 'viswiz')}`;\n

      const height = Math.max(560, Math.min(1100, 440 + nodes.length * 4));\n
      const svg = svgEl('svg', { viewBox: `0 0 ${width} ${height}`, class: 'viswiz-graph-svg', role: 'img', 'aria-label': __('Node graph', 'viswiz') });\n
      activeSvg = svg;\n
      baseView = { x: 0, y: 0, w: width, h: height };\n
      if (!view) view = { ...baseView };\n
      applyView();\n
\n
      const defs = svgEl('defs');\n

      svg.appendChild(defs);\n
      const graphLayer = svgEl('g', { class: 'viswiz-graph-layer' });\n
      svg.appendChild(graphLayer);\n
\n
      const layout = graphLayout(nodes, relations, spec.renderer, width, height);\n
      const nodeMap = new Map(nodes.map((n) => [n.uuid, n]));\n
